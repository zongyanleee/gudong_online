🛡️Firebase Security Rules 防作弊配置規範

本配置確保客戶端只能讀取自身的私有角色資料（例如 users/{myUid} 或 rooms/{id}/private/{uid}）與全場公開的 publicState 節點，嚴防玩家透過 F12 監聽封包竊取隱藏身分。 

// === Cloud Firestore Security Rules (firestore.rules) ===
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // 公開房間節點：供全體玩家監聽大廳座位、票數與發言
    match /GudongOnlineGame/{roomId} {
      allow read: if true;
      allow write: if request.auth != null || request.resource.data.keys().hasAll(['status']);
      
      // 角色隔離私有子集合：僅本人 UID 具備讀取權限 (嚴防 F12 窺探)
      match /privateRoles/{userId} {
        allow read: if request.auth != null && request.auth.uid == userId;
        allow write: if request.auth != null && request.auth.uid == userId;
      }
    }
    
    // 個人私有資料節點
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
  }
}

// === Realtime Database Rules (database.rules.json) ===
{
  "rules": {
    "rooms": {
      "$roomId": {
        "publicState": { ".read": true, ".write": "auth != null" },
        "private": {
          "$uid": {
            ".read": "auth != null && auth.uid === $uid",
            ".write": "auth != null && auth.uid === $uid"
          }
        }
      }
    },
    "users": {
      "$uid": {
        ".read": "auth != null && auth.uid === $uid",
        ".write": "auth != null && auth.uid === $uid"
      }
    }
  }
}
